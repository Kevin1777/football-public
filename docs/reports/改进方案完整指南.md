```python
# 在 app.py 中添加定时任务
from apscheduler.schedulers.background import BackgroundScheduler

def collect_pre_match_data():
    """赛前5分钟采集数据"""
    now = datetime.now()
    upcoming_matches = get_upcoming_matches(minutes=5)

    for match in upcoming_matches:
        # 采集赛前最后的数据
        data = collect_match_data(match)

        # 🔥 重要：不在文件名中包含比分
        filename = f"{match['date']}_{match['league']}_{match['home']}_vs_{match['away']}_{match['time']}.json"
        save_data(data, filename)

        # 比分单独记录
        save_score(match['id'], None)  # 赛前没有比分
```

#### 2.2 完场后更新比分

**分离比分和数据**：

```python
# 完场后，更新比分到单独文件
def update_match_result(match_id, home_score, away_score):
    """完场后更新比分"""
    results_file = "match_results.json"

    results = load_json(results_file)
    results[match_id] = {
        'home_score': home_score,
        'away_score': away_score,
        'updated_at': datetime.now().isoformat()
    }
    save_json(results, results_file)

```

```
5. 完场后2小时：采集结果 → 验证准确率

```
