def _smartsafe_check(self, spm: Dict) -> bool:
    """加强结构风控"""
    structure = spm.get('structure')
    confidence = spm.get('confidence', 0)

    # 🔥 诱导结构：直接拒绝
    if structure == '诱导结构':
        return True  # 触发风控

    # 🔥 中性结构 + 低置信度：拒绝
    if structure == '中性结构' and confidence < 0.6:

    return False

#### 2.2 增强 LVC（凯利指数风控）

```python
def _lvc_check(self, metrics: Dict) -> bool:
    """加强凯利指数风控"""
    kelly_home = abs(metrics.get('kelly_home', 0))
    kelly_away = abs(metrics.get('kelly_away', 0))

    # 🔥 凯利指数异常（>1.5或<0.5）
    if max(kelly_home, kelly_away) > 1.5:
        return True  # 市场过热

    if min(kelly_home, kelly_away) < 0.5:
        return True  # 价值失衡

    return False
```

def _calculate_data_quality_score(self, metrics: Dict) -> float:
    """计算数据质量评分（0-1）"""
    score = 1.0

    # 🔥 检查关键字段
    if metrics.get('total_volume', 0) == 0:
        score -= 0.3  # 无成交量

    if metrics.get('home_pct', 0) + metrics.get('away_pct', 0) < 50:
        score -= 0.2  # 成交分布异常

    if metrics.get('handicap_current', 0) == 0:
        score -= 0.2  # 无盘口数据

    # 欧赔检查
    if metrics.get('home_euro_odds', 0) == 0:
        score -= 0.1

    # 必发指数检查
    if metrics.get('bfindex_home', 0) == 0:
        score -= 0.1

    return max(score, 0.0)

# 使用
data_quality = self._calculate_data_quality_score(metrics)

```python
def _check_data_freshness(self, trade_list: List) -> bool:
    """检查数据时效性"""
    if not trade_list:
        return False

    # 🔥 要求至少有10条交易记录
    if len(trade_list) < 10:
        return False

    # 🔥 检查数据时间跨度（应覆盖至少1小时）
    first_time = self._parse_trade_time(trade_list[0])
    last_time = self._parse_trade_time(trade_list[-1])

    if first_time and last_time:
        time_span = (last_time - first_time).total_seconds() / 3600
        if time_span < 1.0:  # 少于1小时
            return False

    return True
```

## ⭐⭐⭐⭐ 策略 4：联赛分级策略（差异化处理）

### 理念

不同联赛特点不同，应该差异化处理。

### 实现

```python
# 联赛等级配置
LEAGUE_TIERS = {
    'S级': ['英超', '西甲', '德甲', '意甲', '法甲', '欧冠', '世界杯'],
    'A级': ['英冠', '荷甲', '葡超', '巴甲', '中超', '亚冠'],
    'B级': ['英甲', '德乙', '法乙', 'J联赛'],
    'C级': ['其他']
}

def _get_league_tier(self, league: str) -> str:
    """获取联赛等级"""
    for tier, leagues in LEAGUE_TIERS.items():
        if any(l in league for l in leagues):
            return tier
    return 'C级'

def _adjust_confidence_by_league(self, confidence: float, league: str) -> float:
    """根据联赛等级调整置信度"""
    tier = self._get_league_tier(league)

    adjustments = {
        'S级': 1.0,   # 不调整
        'A级': 0.95,  # 略降
        'B级': 0.90,  # 降低
        'C级': 0.80   # 大幅降低
    }

    return confidence * adjustments.get(tier, 0.8)
```

### 策略

```python
# 🔥 差异化阈值
MIN_CONFIDENCE_BY_TIER = {
    'S级': 0.70,  # 顶级联赛：要求低一些

    'B级': 0.80,  # 低级别：要求高

# 应用

min_conf = MIN_CONFIDENCE_BY_TIER[tier]

import xgboost as xgb
from sklearn.model_selection import train_test_split

def train_ml_model():
    """训练XGBoost分类器"""

    # 加载历史数据
    X, y = load_historical_data()  # X=特征，y=实际结果

    # 训练
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2)

    model = xgb.XGBClassifier(
        max_depth=6,
        learning_rate=0.05,

        num_class=3  # home/away/draw

    model.fit(X_train, y_train)

    # 验证
    accuracy = model.score(X_test, y_test)
    print(f"模型准确率: {accuracy:.1%}")

# 集成到推荐引擎
def _ml_enhanced_recommendation(self, features: Dict) -> Dict:
    """ML增强推荐"""

    # X1规则推荐
    x1_result = self._original_recommend(features)
    x1_direction = x1_result['direction']
    x1_confidence = x1_result['confidence']

    # ML模型预测
    ml_proba = self.ml_model.predict_proba([features])[0]
    ml_direction = ['home', 'draw', 'away'][ml_proba.argmax()]
    ml_confidence = ml_proba.max()

    # 🔥 两者一致时提升置信度
    if x1_direction == ml_direction:
        final_confidence = min(x1_confidence * 1.2, 0.95)
        return {'direction': x1_direction, 'confidence': final_confidence}

    # 🔥 不一致时降低置信度
    else:
        final_confidence = x1_confidence * 0.8
        if final_confidence < 0.7:
            return {'direction': 'none'}
        return {'direction': x1_direction, 'confidence': final_confidence}
```
