    url = "https://www.bifaw.com/bifaw/serveodds.php"
    params = {
        "matchid": match_id,
        "marketid": market_id,
        "home_sid": home_sid,
        "away_sid": away_sid,
        "oddstype": "5",  # 5=让球盘
        "action": "result",  # 或其他
    }
